# program test: TestProgram_AnEditCardAppliesTheEdit,
#   TestProgram_AFoldedRunOfReadsOpensInPlace,
#   TestProgram_AClickOnAStripGlyphMovesTheCursor,
#   TestProgram_TheQuitChordArmsThenLeaves
#
# Each step a card (docs/interface/surfaces.md#the-step), through the built
# binary. One turn of four steps: reads only, reads then a command, a write,
# and a command that fails. Read the capture at 04 against the `Cards`
# artboard's whole turn: a padding row inside the band above and below each
# card, one blank between cards; on each header the glyph by precedence, the
# rail on every card that ran or wrote, the verb in body grey and the receipt
# dim, the answer on the right; the sentence that titled the step under it at
# the body column; a footer only where the step has evidence — the command's
# first line out, the write's first line, the failure's line — and none under
# the step that only read. The answer stands on bare screen under the cards.
#
# Then reading mode: the cursor stops on a card as a whole, the bar says what
# enter does to it, and enter opens the large step onto its calls — the reads
# under `▾ read`, rolled up by directory, the two commands under `▾ ran`, each
# with its rail, and the strip under them. Read 07 against the `Cards`
# artboard's large step open (screens 5 and 6). The arrows put a cursor on
# the strip at its last call, the failed command, and walk it back one call
# and on again, its glyph lit and its row lit with it, the bar naming the
# strip's keys (read 08 and 10 against screen 11); enter opens that call's
# own view, esc comes back to the strip where it was, and esc again closes
# the card. Then enter opens it,
# folds it to its header with `▸` in the pointer column, and gives the card
# back.
setup mkdir -p docs internal/ui && printf 'Scenes drive the built binary.\n' > docs/scenes.md
setup for f in keys render reply click fold; do printf 'package ui\n' > internal/ui/$f.go; done
setup printf 'notes nobody has committed\n' > notes.md
setup printf 'iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mP8z8BQDwAEhQGAhKmMIQAAAABJRU5ErkJggg==' | base64 -d > shot.png
snap 01-start "Some things worth doing first"
keys "write the copy story" Enter
# The write and the failing command are put to the reader; the reads and
# the status check run without a card. Each decision waits out the card's
# grace window before it is answered.
snap 02-write-card "[y] apply the change"
sleep 1
keys y
snap 03-command-card "[y] run it once" "Running the suite once more"
sleep 1
keys y
# At 80 columns and up every header keeps its whole receipt; below, the
# second card's command behind its read is counted rather than named, which
# is the narrow rollup (docs/interface/departures.md#the-narrow-rollup-counts-what-the-wide-one-names).
wide 80 snap 04-cards "The story is written" "⚙ read 2 files" "ran git status --short notes.md" "wrote story.md" "?? notes.md" "+ # A code block is copied by itself" "--- FAIL: TestReplyGolden" "exit 1"
narrow 80 snap 04-cards "The story is written" "⚙ read 2 files" "read docs/scenes.md · ran 1 command" "wrote story.md" "?? notes.md" "+ # A code block is copied by itself" "--- FAIL: TestReplyGolden" "exit 1"
keys C-o
# The cursor opens on the turn's close; up twice is the last card, past the
# answer standing between them.
snap 05-reading "[enter] review turn"
keys k
keys k
snap 06-on-the-card "[enter] open it" "Running the suite once more"
keys Enter
# Below 80 columns the bar sheds enter's words before collapse's.
wide 80 snap 07-open "[enter] fold it" "▾ read " "▾ ran 2 commands" "▎✗ printf"
narrow 80 snap 07-open "[-] collapse" "▾ read " "▾ ran 2 commands"
# The first arrow puts the strip's cursor on the last call, the failed
# command; the next two walk it back one call and on again.
keys Left
snap 08-strip "tool 8 of 8" "[←→] along" "❯    ▎✗ printf" "⚙⚙⚙⚙⚙⚙$✗"
keys Left
snap 09-strip-back "tool 7 of 8" "❯    ▎$ git status --short notes.md"
keys Right
snap 10-strip-on-the-failure "tool 8 of 8" "[←→] along" "[enter] open" "❯    ▎✗ printf"
keys Enter
snap 11-the-call "$ printf" "--- FAIL: TestReplyGolden"
keys Escape
snap 12-back-on-the-strip "tool 8 of 8" "❯    ▎✗ printf"
keys Escape
snap 13-closed "[enter] open it" "Running the suite once more"
keys Enter
wide 80 snap 14-open-again "[enter] fold it" "▾ ran 2 commands"
narrow 80 snap 14-open-again "[-] collapse" "▾ ran 2 commands"
keys Enter
snap 15-folded "[enter] unfold it" "❯▎✗ read"
# The cursor stands in the pointer column, so the fold's own mark shows once
# it moves off the card.
keys k
snap 16-folded-mark "▸▎✗ read"
keys j
keys Enter
snap 17-card-again "[enter] open it" "Running the suite once more"
keys q
snap 18-prompt "╭─ ✓ done"
# A read of a picture: the card's footer row names it and says it opens like
# an attachment; the reading cursor stops on the row, and enter opens the
# attachment card the way a picture the reader sent opens.
keys "look at the capture" Enter

narrow 80 snap 19-picture "That is the capture" "▣ shot.png · 1×1"
keys C-o
sleep 1
keys k
keys k
sleep 1
snap 20-on-the-picture "❯   ▣ shot.png"
keys Enter
snap 21-attachment-card "shot.png" "1×1"
keys Escape
sleep 1
keys q
snap 22-prompt "╭─ ✓ done"
keys C-c
snap 23-quit-armed "again quits"
keys C-c
