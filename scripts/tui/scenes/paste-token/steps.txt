# program test: TestProgram_APasteTooBigForTheDraftIsAToken
#
# A log too big for the draft, pasted into the middle of a sentence: the fold
# it leaves, what the vitals say it will cost, the reader its chip opens onto,
# and the row the send leaves behind (docs/interface/surfaces.md#the-input-frame).
# The chip is reached through reading mode, which keeps the sentence: with
# nothing in the transcript yet the mode opens on the staged strip, and once a
# turn is there j off its last row lands on it
# (docs/interface/surfaces.md#a-staged-attachment).
#
# The paste is a real bracketed paste of a file rather than typed keystrokes.
# Typing it would be two hundred lines of key presses, which is the other
# question entirely: what the surface does with text arriving all at once is
# what the threshold is about.
setup printf '=== RUN   TestRoundLimit\n' > loop.log
setup for i in $(seq 1 211); do printf '    loop_test.go:44: round %d reached after 2.1s, loop still running with 3 goroutines parked on the same channel\n' "$i" >> loop.log; done
setup printf -- '--- FAIL: TestRoundLimit (2.11s)\nexit status 1\n' >> loop.log
snap 01-start "Some things worth doing first"
keys "this test log says the loop never stops "
paste loop.log
snap 02-fold "will cost" "⟨Paste#1 · 214 lines⟩"
keys " fix the exit condition"
snap 03-priced "fix the exit condition" "Paste#1 will cost"
keys C-o
snap 03b-strip "[enter] open" "back to the draft"
keys Enter
snap 04-open "PASTE#1 · lines"
keys q
snap 05-back "[enter] open" "back to the draft"
keys Escape
snap 05b-draft "[enter] send" "fix the exit condition"
keys Enter
snap 06-sent "paste-1.txt · 214 lines" "⟨¶ Paste#1⟩"
# ↑ on the sent sentence. The send emptied the staging area, so what the fold
# stands for has to come back with the words: it is staged again from the row's
# own bytes, priced on the vitals as it was before the send, and opened by its
# chip again — rather than five words about a log that is not riding (recall.go).
keys Up
snap 07-recalled "Paste#1 will cost" "⟨Paste#1 · 214 lines⟩ fix the exit condition"
keys C-o
keys j
snap 07b-strip "[enter] open" "back to the draft"
keys Enter
snap 08-reopened "PASTE#1 · lines"
keys q
keys C-c
keys C-c
