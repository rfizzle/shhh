# program test: none — the screen's host is the command package's config
#   model, which the chat program test does not build; the header word is held
#   by TestConfigScreen_AnOutdatedFileSaysSo and drawn in the config-screen-scope
#   golden's third panel
#
# /config in a session standing in a checkout, over a person's own settings
# file that is older than the table. The harness's config.toml lists no
# commented row, and a file that lists none is never outdated (KeysBehind
# counts nothing for it), so the setup below appends one commented row to the
# file under the scene's home, which lists that one key and leaves every
# other out. The checkout's own file has no such row and says nothing; moving
# the write to the person's own file (shift+tab) draws the header
# `/config · your file · ~/.config/shhh/config.toml · outdated`, the word in
# the warn colour after the path.
setup printf '# max_tool_rounds = 150\n' >> "$home/config/shhh/config.toml"
snap 01-start "Some things worth doing first"
keys "/config" Enter
snap 02-checkout "/config · the checkout's file · .shhh/config.toml" "FLOWS"
keys BTab
snap 03-outdated "/config · your file" "config.toml · outdated"
keys Escape
sleep 0.3
keys Escape
sleep 0.5
