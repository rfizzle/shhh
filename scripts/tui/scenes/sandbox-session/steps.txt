# program test: TestProgram_ARunningCommandTailsIntoItsRow, TestProgram_TheQuitChordArmsThenLeaves
#
# golden: TestGolden_SandboxSession (sandbox-session); the container behind
# the screen is internal/cli's TestSandboxContainmentValues, and a writer's
# own container is TestSandboxWriterGetsItsOwnContainer.
#
# A session whose commands run in a disposable container, driven on a host
# with no engine: a stand-in `docker` (this directory) answers the create,
# the probe, an exec and the remove, and runs an exec's argv here under the
# built shhh as the helper, so the helper's flags and its stream are real.
# Read the captures against the golden's layout block: /status names the
# container sandbox and its profile, and says a writer's commands are
# contained by the container sandbox, in a container of the writer's own; the
# command's three lines arrive a second apart through the
# helper, and the row settles with all three. A second turn starts a process,
# which is one more exec through the helper with the stream as its stdin, and
# reads a line from it and stops it: the card is the ordinary start card (the
# mechanism is the container's), the read answers "ready" from the process's
# own stdout, and the stop signals the exec's client, whose hang-up is the
# helper's cue to stop the process. The quit removes the container, which the
# stand-in does not show and the program tests do.
#
# Then a command the person types. `!echo hi` is theirs, and in a sandbox
# session it runs in the container too: the stand-in's exec answers it through
# the helper like the assistant's, and the row ends in `container` to say where
# it ran (golden: TestGolden_SandboxBang). A host session's row says nothing of
# where; this one does, and the run carries no ceiling. The row reads
# `ok · container · 1 line` beside `echo hi`.
setup mkdir -p ../engine && cp "$scene/docker" ../engine/docker && chmod +x ../engine/docker
snap 01-start "Some things worth doing first"
keys "/status" Enter
snap 02-status "container sandbox ·" "Containment" "a writer's commands: contained · container sandbox"
keys "count to three" Enter
snap 03-card "[y] run it once"
keys y
snap 04-tail "two" "▸ run"
snap 05-settled "The three lines came back from the container" "three"
keys "!echo hi" Enter
snap 06-bang-card "[y] run it once"
keys y
snap 07-bang-row "echo hi" "ok · container"
keys "start a ticker process" Enter
snap 08-process-card "[y] allow it" "ticker"
keys y
snap 09-process-read "The process printed ready inside the container"
# Then a writer (golden: TestGolden_SandboxWriterSpawnCard). Its card says,
# before the answer, what its commands run under: `commands  contained ·
# container sandbox`, the session indicator's words. Started, the writer is
# given a copy of the checkout and a container of its own whose one mount is
# that copy — the stand-in's second create — and its one command is put to
# you on its own card, named for the writer. Answered, it runs in the
# writer's container through the helper, and the writer's report says what
# it printed. The quit removes the writer's container with the session's;
# the stand-in writes both names to its log of removals, and the screen does
# not show it (internal/cli's TestSandboxWriterGetsItsOwnContainer does).
keys "start a writer" Enter
snap 10-writer-card "Start a writer" "contained · container sandbox"
sleep 1
keys y
snap 11-writer-ask "echo from-the-writer" "writer-1"
sleep 1
# The writer's card arrives over an idle draft, which keeps the keyboard:
# ctrl+space hands it to the card, then y allows the command.
keys C-Space
keys y
snap 12-writer-done "✓ done" "writer-1"
keys C-c
snap 13-quit-armed "again quits"
keys C-c
snap 14-exit "that is everything the screen was holding"
