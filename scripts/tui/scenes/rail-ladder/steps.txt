# program test: TestProgram_TheRailsFootLineOpensTheWholeRail
#
# The rail on a short terminal gives up the least alive thing first, and what
# it drops it still counts on a foot line that is a door
# (docs/interface/surfaces.md#the-inspector-rail). The session sends one
# writer and goes on thinking for a minute and a half, so its turn runs for the
# whole scene; the writer asks to write a file, and the card for it is left
# standing over the draft, so the writer waits on the reader until the scene's
# last steps answer it.
#
# The request is an edit rather than a command on purpose: a child's command
# card needs a containment mechanism, and on a host without one the command is
# refused before any card is drawn (rail-agents-preset says more).
#
# A click is the terminal's own SGR report, a press and a release in one cell,
# counted from 1. At 130 columns the rail's first column is 85; the foot line's
# row depends on how tall the card standing over the draft is, so the click
# reads it off the pane: the first row carrying `↓ N more ·`, at column 90.
#
# What the capture is for, at 130 columns where the rail is on screen: at 04
# the rail is twenty-two rows' worth — the turn and the map ask, so they keep
# their place and collapse to their headings, and the static blocks have left
# for the foot line on the rail's last row, which names them; at 05 a click on
# that line has put the rail in full over the pane, every block with nothing
# shed; at 06 a click on the same cell has put the short rail back as it was;
# at 07 /rail, the key twin, opens the same screen and at 08 its own esc
# leaves it; at 09 the writer, let through, has written and asks to land its
# patch on a card of its own; at 10 that is answered too, the writer has
# finished, the map is history, and the rail has its rows back in the same
# order. Below 130 there
# is no rail: 04 waits on the writer's row in the fan-out, the clicks are not
# sent, and /rail still opens the rail in full, which is how a narrow terminal
# reads it.
setup printf 'package agent\n\nfunc loop() {\n    for round := 0; round < limit; round++ {\n        step()\n    }\n}\n' > loop.go
snap 01-start "Some things worth doing first"
keys "survey the round accounting" Enter
# Spawning is gated: the answer waits out the card's grace window.
snap 02-spawn-card "Start a writer"
sleep 1
keys y
sleep 3
# writer-1's edit is routed here as a card. esc hands the keyboard back to the
# draft and denies nothing: the card stays standing over the draft and the
# writer goes on waiting on the reader.
snap 03-edit-card "writer-1 ▸ Approve edit"
sleep 1
keys Escape
sleep 2
wide 130 snap 04-short-rail "more ·" "THIS TURN"
narrow 130 snap 04-short-rail "writer-1"
# The foot line's cell, read off the pane.
wide 130 keys -l "$(r=$(tmux -L "$SOCK" capture-pane -p -t scene | grep -n -m1 '↓ [0-9]* more ·' | cut -d: -f1); printf '\033[<0;90;%sM\033[<0;90;%sm' "$r" "$r")"
wide 130 snap 05-rail-open "/rail · the rail in full" "nothing shed"
wide 130 keys -l "$(r=$(tmux -L "$SOCK" capture-pane -p -t scene | grep -n -m1 '↓ [0-9]* more ·' | cut -d: -f1); printf '\033[<0;90;%sM\033[<0;90;%sm' "$r" "$r")"
wide 130 snap 06-rail-closed "more ·" "THIS TURN"
# The key twin, from the draft, at every width.
keys "/rail" Enter
snap 07-rail-command "CONTEXT" "THIS TURN" "AGENTS"
keys Escape
sleep 1
wide 130 snap 08-rail-left "more ·" "THIS TURN"
narrow 130 snap 08-rail-left "writer-1"
# The writer's request answered: the handover gives the card the keyboard and
# [y] lets the edit through, and the writer finishes on its report.
keys C-Space
sleep 1
keys y
# The writer writes and then asks to land its patch: the second card is a
# decision of its own, drawn full width with the rail put away, and [y]
# answers it from where the keyboard already is.
snap 09-patch-card "writer-1 ▸ Apply patch"
keys y
sleep 2
wide 130 snap 10-answered "AGENTS" "THIS TURN"
narrow 130 snap 10-answered "writer-1"
keys C-c
sleep 0.6
keys C-c
sleep 0.6
keys C-c
# The quit offers a handoff over the writer's uncommitted file; n declines.
snap 11-handoff-offer "Write a handoff first?" "Changes are uncommitted.  [y/N]"
keys n
snap 12-exit "that is everything the screen was holding"
