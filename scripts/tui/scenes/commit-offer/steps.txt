# program test: TestProgram_TheCloseOffersTheCommitAndTheCardCommits,
#   TestProgram_ASelectedCloseIsCommittedThroughTheHandover,
#   TestProgram_TheQuitChordArmsThenLeaves
#
# A turn's close offers no key for taking it back or committing it until it
# is selected, and then only review and the commit card the handover opens
# (docs/interface/surfaces.md#the-turns-close); taking it back is /undo,
# which the row's note names. The commit itself is asked of the model, and it
# arrives as the write tool's cards. The workspace is a repository with one
# commit in it, one tracked file the turn edits, and one the reader changed by
# hand and never staged — which is what the card's `leaves` row is read for:
# the commit carries the turn's own work and nothing of theirs. The turn
# also writes a token into a new settings file, which is what the card's
# `secrets` row and its refusal are read for
# (docs/capabilities/secrets.md#a-secret-does-not-get-committed).
#
# The identity is repository-local because the pane runs under a home of its
# own, so there is no global one for the commit to find.
setup printf 'package agent\n\nconst limit = 25\n' > loop.go
setup printf '# project\n' > README.md
setup git add loop.go README.md
setup git config user.email tui@shhh
setup git config user.name tui
setup git commit -q -m 'seed the repository'
setup printf '# project\n\nnotes of my own\n' > README.md
snap 01-start "Some things worth doing first"
keys "cap rounds at the limit instead of erroring" Enter
# The edit is gated, so the card is what the turn stops on. The answer waits
# out the card's grace window and then hands the keyboard over before
# pressing: a letter that arrives while the draft still holds the keyboard
# goes into the draft.
snap 02-edit-card "const limit = 50"
sleep 1
keys C-Space
sleep 0.3
keys y
# The settings file is a write of its own, and so a card of its own.
# It adds a token on line 12, and the card says so before it lands
# (docs/capabilities/approvals-and-safety.md#a-write-that-adds-a-secret-is-always-asked).
snap 02b-write-card "config/dev.env" "adds 1 github token · line 12"
sleep 1
keys C-Space
sleep 0.3
keys y
# The close states what changed and offers nothing while nothing selects it;
# its note names /undo as the way back where the row has room for it.
snap 03-close "2 files changed" "The rounds are capped at the limit now"
# The pointer selects it from the prompt, mid-sentence: the row then draws
# review and the commit, both chords, so neither is a letter of the sentence.
keys "and while you are at it"
sleep 0.3
keys S-Up
snap 04-selected "[enter] review turn" "[ctrl+space/ctrl+y] commit"
# The handover opens the card with the sentence still in the box behind it.
keys C-Space
snap 05-card "Commit this turn" "README.md changed by you, never staged" "[s] read the hunks" "1 github token · config/dev.env:12" "[!] commit with the secret — it stays in history"
# The token the turn wrote is named by kind, file and line, and enter is
# refused while it stands: nothing is staged, and the card says what to do.
sleep 0.5
keys Enter
snap 05b-refused "a secret is in the diff — remove it, or commit it on purpose with [!]"
sleep 0.5
keys Escape
sleep 0.5
# Esc leaves the changeset standing, and the draft with it; the second one
# drops the pointer and the third clears the box, so what follows is the
# commit asked of the model instead.
keys Escape
sleep 0.3
keys Escape
sleep 0.3
keys "commit that change" Enter
# Staging is a write to the repository, so it is a card of its own, and so
# is the commit after it. Each card draws the write tool's own fields — what
# is staged, that nothing is pushed, the hooks and the way back — and never
# a shell's reading of the git line the deny list matches.
snap 06-add-card "$ stage 1 file" "this session's files only"
sleep 1
keys C-Space
sleep 0.3
keys y
snap 07-commit-card "$ commit" "hooks" "git revert"
sleep 1
keys C-Space
sleep 0.3
keys y
# The receipt is the row the close draws for a commit, and it names what an
# undo does not reach.
snap 08-receipt "Committed loop.go, and nothing of yours" "committed 1 file as" "/undo does not reach history"
# The draft is empty, so the first cancel arms the quit and the second leaves.
keys C-c
snap 09-quit-armed "again quits"
keys C-c
# The quit offers a handoff over uncommitted work; n declines.
snap 10a-handoff-offer "Write a handoff first?" "Changes are uncommitted.  [y/N]"
keys n
snap 10-exit "that is everything the screen was holding"
