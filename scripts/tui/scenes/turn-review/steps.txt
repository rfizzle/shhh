# program test: TestProgram_EnterOnASelectedCloseOpensThatTurnsReview,
#   TestProgram_AClickOnACloseOpensThatTurnsReview,
#   TestProgram_TheQuitChordArmsThenLeaves,
#   TestProgram_TurnsOpensEveryTurnAndReviewsOne
#
# A turn's changed-files row is the door to that turn's review
# (docs/interface/surfaces.md#the-turns-close). Two turns close on the same
# words, and nothing on either offers a key until it is selected: the pointer
# selects the newer from the prompt, its row says what enter does there, and
# enter opens turn 2. Then a click on the older one's row opens turn 1 — the
# row clicked, not the newest — from a half-typed line that is still in the
# box once the review is left.
#
# Between the two, /turns puts both turns in one list, newest first, the
# close of the one under the pointer beside it; enter on the turn just
# reviewed opens that review again, and its esc comes back to the list
# (docs/interface/surfaces.md#the-supporting-screens).
#
# Driven at 60 columns, the narrowest the close is drawn at, and read at 130,
# where the list and the close stand side by side.
setup printf 'package agent\n\nconst limit = 25\n' > loop.go
setup git add loop.go
setup git config user.email tui@shhh
setup git config user.name tui
setup git commit -q -m 'seed the repository'
snap 01-start "Some things worth doing first"
keys "cap rounds at the limit" Enter
# The edit is gated; the answer waits out the card's grace window and hands
# the keyboard over before pressing.
snap 02-first-card "const limit = 50"
sleep 1
keys C-Space
sleep 0.3
keys y
snap 03-first-close "Capped at fifty now" "1 file changed"
keys "raise the cap" Enter
snap 04-second-card "const limit = 100"
sleep 1
keys C-Space
sleep 0.3
keys y
# Two closes, each stating what its turn changed, and neither offering a key
# until it is selected. The session's first reading
# of its run landed in turn 1 and is quiet — an unclear verdict the rail would
# draw whole — so it leaves no row, and the layout the click below is aimed at
# is the two turns alone.
snap 05-two-closes "Raised again" "✓ Done · 1 tool"
# The pointer's first press lands on the newest close, which then draws its
# offers as the selected row's.
keys S-Up
snap 06-selected "[enter] review turn" "] commit"
keys Enter
snap 07-review-turn-2 "leave, change nothing" "REVIEW turn 2" "const limit = 100"
keys Escape
sleep 0.5
keys Escape
sleep 0.5
# Every turn the session has run, newest first: turn 2 under the pointer, its
# close drawn whole, and the review of it the one offer that names it.
keys "/turns" Enter
snap 08-turns "[enter] review turn 2" "✓ turn 2" "✓ turn 1" "1 file changed"
keys Enter
snap 09-turns-review "leave, change nothing" "REVIEW turn 2" "const limit = 100"
# The review's way out is back to the list the reader was walking.
keys Escape
snap 10-turns-back "[enter] review turn 2" "✓ turn 1"
keys Escape
sleep 0.5
# A half-typed line, then a click on the older close's changed-files row —
# row 14 at this size — as the SGR press and release a terminal reports. The
# review it opens is turn 1's, the change from 25 to 50.
keys "and then"
snap 11-draft "and then"
keys -l $'\e[<0;12;14M' $'\e[<0;12;14m'
snap 12-review-turn-1 "leave, change nothing" "REVIEW turn 1" "const limit = 25"
keys Escape
snap 13-draft-kept "and then"
keys C-c
sleep 0.3
keys C-c
snap 14-quit-armed "again quits"
keys C-c
snap 15-exit "that is everything the screen was holding"
