# The README's hero and two of its stills: one `shhh code` turn under a
# person's hands, typed at a person's pace — an edit card answered, a command
# card answered, the answer, the turn's close and the commit it offers. It is

# small Go package, a plausible ask, and a placeholder model (its launch).

# it unpaced, so it is also a test that the route still reaches every card.
#
# The identity is repository-local because the pane runs under a home of its
# own, so there is no global one for the commit to find.
setup printf 'package fetch\n\nimport "time"\n\nconst maxAttempts = 3\n\n// Retry calls fn until it succeeds or the attempts run out, doubling the\n// wait between them from 200ms.\nfunc Retry(fn func() error) error {\n\twait := 200 * time.Millisecond\n\tvar err error\n\tfor i := 0; i < maxAttempts; i++ {\n\t\tif err = fn(); err == nil {\n\t\t\treturn nil\n\t\t}\n\t\ttime.Sleep(wait)\n\t\twait *= 2\n\t}\n\treturn err\n}\n' > retry.go
setup printf 'module example.com/fetch\n\ngo 1.24\n' > go.mod
setup git add retry.go go.mod
setup git config user.email you@example.com
setup git config user.name you
setup git commit -q -m 'add a retry helper'
